local function CreateTab(tabName, iconText)
    local page = Instance.new("ScrollingFrame")
    page.Parent = TabContainer
    page.BackgroundTransparency = 1
    page.Size = UDim2.new(1, 0, 1, 0)
    page.CanvasSize = UDim2.new(0, 0, 2.2, 0)
    page.ScrollBarThickness = 2
    page.Visible = false

    local btn = Instance.new("TextButton")
    btn.Parent = Sidebar
    btn.BackgroundColor3 = Color3.fromRGB(18, 18, 24)
    btn.Size = UDim2.new(0.9, 0, 0, 28)
    btn.Position = UDim2.new(0.05, 0, 0, (#TabButtons * 32) + 6)
    btn.Font = Enum.Font.GothamBold
    btn.Text = iconText .. " " .. tabName
    btn.TextColor3 = Color3.fromRGB(130, 130, 150)
    btn.TextSize = 10
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 4)

    btn.MouseButton1Click:Connect(function()
        for _, p in pairs(Tabs) do p.Visible = false end
        for _, b in pairs(TabButtons) do 
            b.BackgroundColor3 = Color3.fromRGB(18, 18, 24)
            b.TextColor3 = Color3.fromRGB(130, 130, 150)
        end
        page.Visible = true
        btn.BackgroundColor3 = Color3.fromRGB(0, 120, 215)
        btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    end)

    table.insert(TabButtons, btn)
    Tabs[tabName] = page
    return page
end

local function AddToggle(parent, text, defaultVal, yPos, cb)
    local btn = Instance.new("TextButton")
    btn.Parent = parent
    btn.BackgroundColor3 = Color3.fromRGB(22, 22, 30)
    btn.Position = UDim2.new(0.02, 0, 0, yPos)
    btn.Size = UDim2.new(0.96, 0, 0, 26)
    btn.Font = Enum.Font.GothamMedium
    btn.Text = "  " .. text
    btn.TextColor3 = Color3.fromRGB(210, 210, 225)
    btn.TextSize = 10
    btn.TextXAlignment = Enum.TextXAlignment.Left
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 4)

    local stateLbl = Instance.new("TextLabel")
    stateLbl.Parent = btn
    stateLbl.BackgroundTransparency = 1
    stateLbl.Position = UDim2.new(1, -55, 0, 0)
    stateLbl.Size = UDim2.new(0, 50, 1, 0)
    stateLbl.Font = Enum.Font.GothamBold
    stateLbl.Text = defaultVal and "ON" or "OFF"
    stateLbl.TextColor3 = defaultVal and Color3.fromRGB(0, 255, 150) or Color3.fromRGB(255, 70, 70)
    stateLbl.TextSize = 10

    local val = defaultVal
    btn.MouseButton1Click:Connect(function()
        val = not val
        stateLbl.Text = val and "ON" or "OFF"
        stateLbl.TextColor3 = val and Color3.fromRGB(0, 255, 150) or Color3.fromRGB(255, 70, 70)
        cb(val)
    end)
end

local function AddDropdown(parent, title, options, yPos, cb)
    local lbl = Instance.new("TextLabel")
    lbl.Parent = parent
    lbl.BackgroundTransparency = 1
    lbl.Position = UDim2.new(0.02, 0, 0, yPos)
    lbl.Size = UDim2.new(0.96, 0, 0, 18)
    lbl.Font = Enum.Font.GothamBold
    lbl.Text = title .. ": [ " .. options[1] .. " ]"
    lbl.TextColor3 = Color3.fromRGB(255, 170, 0)
    lbl.TextSize = 10
    lbl.TextXAlignment = Enum.TextXAlignment.Left

    local idx = 1
    local btn = Instance.new("TextButton")
    btn.Parent = parent
    btn.BackgroundColor3 = Color3.fromRGB(28, 28, 38)
    btn.Position = UDim2.new(0.02, 0, 0, yPos + 20)
    btn.Size = UDim2.new(0.96, 0, 0, 24)
    btn.Font = Enum.Font.GothamMedium
    btn.Text = "Chạm để đổi vị trí"
    btn.TextColor3 = Color3.fromRGB(220, 220, 230)
    btn.TextSize = 10
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 4)

    btn.MouseButton1Click:Connect(function()
        idx = idx + 1
        if idx > #options then idx = 1 end
        lbl.Text = title .. ": [ " .. options[idx] .. " ]"
        cb(options[idx])
    end)
end

local function AddSlider(parent, text, minVal, maxVal, defaultVal, yPos, cb)
    local lbl = Instance.new("TextLabel")
    lbl.Parent = parent
    lbl.BackgroundTransparency = 1
    lbl.Position = UDim2.new(0.02, 0, 0, yPos)
    lbl.Size = UDim2.new(0.6, 0, 0, 24)
    lbl.Font = Enum.Font.GothamMedium
    lbl.Text = "  " .. text
    lbl.TextColor3 = Color3.fromRGB(190, 190, 205)
    lbl.TextSize = 10
    lbl.TextXAlignment = Enum.TextXAlignment.Left

    local box = Instance.new("TextBox")
    box.Parent = parent
    box.BackgroundColor3 = Color3.fromRGB(25, 25, 35)
    box.Position = UDim2.new(0.65, 0, 0, yPos)
    box.Size = UDim2.new(0.33, 0, 0, 24)
    box.Font = Enum.Font.GothamBold
    box.Text = tostring(defaultVal)
    box.TextColor3 = Color3.fromRGB(0, 200, 255)
    box.TextSize = 10
    Instance.new("UICorner", box).CornerRadius = UDim.new(0, 4)

    box.FocusLost:Connect(function()
        local n = tonumber(box.Text)
        if n then
            local clamped = math.clamp(n, minVal, maxVal)
            box.Text = tostring(clamped)
            cb(clamped)
        end
    end)
end

-- HỆ THỐNG XỬ LÝ SCRIPT (AIMBOT + ESP ENGINE)
-- ============================================================================
local ESPRegistry = {}

RunService.RenderStepped:Connect(function()
    -- Xử lý Aimbot
    if Config.AimbotEnabled then
        local targetPartName = "Head"
        if Config.AimTarget == "Neck" then targetPartName = "UpperTorso"
        elseif Config.AimTarget == "Body" or Config.AimTarget == "Chest" then targetPartName = "HumanoidRootPart" end

        local closestPart = nil
        local minDist = Config.FOVRate

        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character and plr.Character:FindFirstChild("Humanoid") and plr.Character.Humanoid.Health > 0 then
                local part = plr.Character:FindFirstChild(targetPartName) or plr.Character:FindFirstChild("Head")
                if part then
                    local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
                    if onScreen then
                        local mPos = UserInputService:GetMouseLocation()
                        local dist = (Vector2.new(screenPos.X, screenPos.Y) - mPos).Magnitude
                        if dist < minDist then
                            minDist = dist
                            closestPart = part
                        end
                    end
                end
            end
        end

        if closestPart then
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, closestPart.Position)
        end
    end

    -- Xử lý ESP
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local char = plr.Character
            if char and char:FindFirstChild("HumanoidRootPart") and char:FindFirstChildOfClass("Humanoid") then
                local hrp = char.HumanoidRootPart
                local hum = char:FindFirstChildOfClass("Humanoid")
                local head = char:FindFirstChild("Head")

                if not ESPRegistry[plr] then
                    ESPRegistry[plr] = {
                        Box = Drawing.new("Square"),
                        Name = Drawing.new("Text"),
                        HP = Drawing.new("Text"),
                        Tracer = Drawing.new("Line"),
                        HeadDot = Drawing.new("Circle")
                    }
                    local d = ESPRegistry[plr]
                    d.Box.Filled = false
                    d.Box.Thickness = 1
                    d.Name.Center = true
                    d.Name.Size = 11
                    d.HP.Center = true
                    d.HP.Size = 10
                    d.HeadDot.Filled = true
                    d.HeadDot.Radius = 4
                    d.HeadDot.Color = Color3.fromRGB(255, 0, 0)
                end

                local d = ESPRegistry[plr]
                local vector, onScreen = Camera:WorldToViewportPoint(hrp.Position)

                if onScreen and hum.Health > 0 and Config.EnemyESP then
                    local boxW = 1500 / vector.Z
                    local boxH = 2600 / vector.Z
                    local boxX = vector.X - boxW / 2
                    local boxY = vector.Y - boxH / 2

                    -- Box
                    if Config.ESP_Box then
                        d.Box.Visible = true
                        d.Box.Color = Color3.fromRGB(0, 200, 255)
                        d.Box.Size = Vector2.new(boxW, boxH)
                        d.Box.Position = Vector2.new(boxX, boxY)
                    else d.Box.Visible = false end

                    -- Name & Distance
                    if Config.ESP_Nickname or Config.ESP_Distance then
                        d.Name.Visible = true
                        local textStr = ""
                        if Config.ESP_Nickname then textStr = textStr .. plr.Name end
                        if Config.ESP_Distance then 
                            local dist = math.floor((LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") and (LocalPlayer.Character.HumanoidRootPart.Position - hrp.Position).Magnitude) or 0)
                            textStr = textStr .. " [" .. dist .. "m]"
                        end
                        d.Name.Text = textStr
                        d.Name.Color = Color3.fromRGB(255, 255, 255)
                        d.Name.Position = Vector2.new(vector.X, boxY - 14)
                    else d.Name.Visible = false end

                    -- Health
                    if Config.ESP_Health then
                        d.HP.Visible = true
                        d.HP.Text = "HP: " .. math.floor(hum.Health)
                        d.HP.Color = Color3.fromRGB(0, 255, 120)
                        d.HP.Position = Vector2.new(vector.X, boxY + boxH + 2)
                    else d.HP.Visible = false end

                    -- Line (Tracer)
                    if Config.ESP_Line then
                        d.Tracer.Visible = true
                        d.Tracer.Color = Color3.fromRGB(0, 200, 255)
                        d.Tracer.From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
                        d.Tracer.To = Vector2.new(vector.X, vector.Y)
                    else d.Tracer.Visible = false end

                    -- Head Red Dot
                    if Config.ESP_HeadDot and head then
                        local headVec, headOnScreen = Camera:WorldToViewportPoint(head.Position)
                        if headOnScreen then
                            d.HeadDot.Visible = true
                            d.HeadDot.Position = Vector2.new(headVec.X, headVec.Y)
                        else d.HeadDot.Visible = false end
                    else d.HeadDot.Visible = false end
                else
                    for _, obj in pairs(d) do obj.Visible = false end
                end
            end
        end
    end
end)
