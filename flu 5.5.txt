    local ESPRegistry = {}

    local function IsEnemy(player)
        if player == LocalPlayer then return false end
        if SelectedTargetPlayer ~= "Tất cả (Mặc định)" and player.Name ~= SelectedTargetPlayer then return false end
        if not Config.TeamCheck then return true end
        if player.Team and LocalPlayer.Team and player.Team == LocalPlayer.Team then return false end
        return true
    end

    local function IsVisible(targetPart)
        if not targetPart then return false end
        local origin = Camera.CFrame.Position
        local direction = targetPart.Position - origin
        local raycastParams = RaycastParams.new()
        raycastParams.FilterType = Enum.RaycastFilterType.Exclude
        local ignoreList = {LocalPlayer.Character}
        if targetPart.Parent then table.insert(ignoreList, targetPart.Parent) end
        raycastParams.FilterDescendantsInstances = ignoreList
        local result = Workspace:Raycast(origin, direction, raycastParams)
        return result == nil
    end

    task.spawn(function()
        while true do
            task.wait(0.2)
            if Config.InfiniteAmmo then
                pcall(function()
                    local char = LocalPlayer.Character
                    if char then
                        local tool = char:FindFirstChildOfClass("Tool")
                        if tool then
                            for _, v in pairs(tool:GetDescendants()) do
                                if v:IsA("NumberValue") or v:IsA("IntValue") then
                                    local nameLower = v.Name:lower()
                                    if nameLower:find("ammo") or nameLower:find("clip") or nameLower:find("bul") then
                                        v.Value = 999
                                    end
                                end
                            end
                        end
                    end
                end)
            end
        end
    end)

    RunService.RenderStepped:Connect(function()
        if Config.ShowFOV then
            FOVCircle.Visible = true
            if Config.LockCrosshairFOV then
                FOVCircle.Position = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
            else
                FOVCircle.Position = UserInputService:GetMouseLocation()
            end
        else
            FOVCircle.Visible = false
        end

        if Config.FixLagMax then
            pcall(function() settings().Rendering.QualityLevel = Enum.QualityLevel.Level01 end)
        end

        if Config.NoClip then
            pcall(function()
                local char = LocalPlayer.Character
                if char then
                    for _, p in pairs(char:GetDescendants()) do if p:IsA("BasePart") then p.CanCollide = false end end
                end
            end)
        end

        if Config.SpeedEnabled then
            pcall(function()
                local char = LocalPlayer.Character
                if char and char:FindFirstChildOfClass("Humanoid") then
                    char.Humanoid.WalkSpeed = Config.WalkSpeed
                end
            end)
        end

        if Config.AimbotEnabled or Config.AimSilentV2 or Config.GhostMode or Config.AimKill then
            pcall(function()
                local closestPart = nil
                local referencePos = Config.LockCrosshairFOV and Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2) or UserInputService:GetMouseLocation()
                local minDist = Config.FOVRate

                for _, plr in ipairs(Players:GetPlayers()) do
                    if IsEnemy(plr) and plr.Character and plr.Character:FindFirstChild("Humanoid") and plr.Character.Humanoid.Health > 0 then
                        local part = plr.Character:FindFirstChild(Config.HitPartTarget ~= "Default" and Config.HitPartTarget or "Head")
                        if part then
                            if IsVisible(part) then
                                local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
                                if onScreen then
                                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - referencePos).Magnitude
                                    if dist < minDist then
                                        minDist = dist
                                        closestPart = part
                                    end
                                end
                            end
                        end
                    end
                end

                if closestPart then
                    if Config.AimbotEnabled or Config.GhostMode or Config.AimKill then
                        Camera.CFrame = Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position, closestPart.Position), Config.AimKill and 1.0 or 0.3)
                    elseif Config.AimSilentV2 then
                        Camera.CFrame = CFrame.new(Camera.CFrame.Position, closestPart.Position)
                    end

                    if Config.AimFire then
                        local char = LocalPlayer.Character
                        if char then
                            local tool = char:FindFirstChildOfClass("Tool")
                            if tool then tool:Activate() end
                        end
                    end
                end
            end)
        end

        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer then
                local char = plr.Character
                if char and char:FindFirstChild("HumanoidRootPart") and char:FindFirstChildOfClass("Humanoid") then
                    local hrp = char.HumanoidRootPart
                    local hum = char:FindFirstChildOfClass("Humanoid")

                    if not ESPRegistry[plr] then
                        ESPRegistry[plr] = {
                            Box2D = Drawing.new("Square"),
                            Box3DLines = {Drawing.new("Line"), Drawing.new("Line"), Drawing.new("Line"), Drawing.new("Line")},
                            SkeletonLines = {
                                Drawing.new("Line"), Drawing.new("Line"), Drawing.new("Line"), 
                                Drawing.new("Line"), Drawing.new("Line"), Drawing.new("Line")
                            },
                            Name = Drawing.new("Text"),
                            HP = Drawing.new("Text"),
                            Tracer = Drawing.new("Line")
                        }
                        local d = ESPRegistry[plr]
                        d.Box2D.Filled = false
                        d.Name.Center = true
                        d.Name.Size = 11
                        d.HP.Center = true
                        d.HP.Size = 10
                    end

                    local d = ESPRegistry[plr]
                    local vector, onScreen = Camera:WorldToViewportPoint(hrp.Position)

                    if onScreen and hum.Health > 0 and Config.EnemyESP then
                        local boxW = 1500 / vector.Z
                        local boxH = 2600 / vector.Z
                        local boxX = vector.X - boxW / 2
                        local boxY = vector.Y - boxH / 2

                        if Config.ESP_Box2D then
                            d.Box2D.Visible = true
                            d.Box2D.Color = Config.ESP_Color
                            d.Box2D.Size = Vector2.new(boxW, boxH)
                            d.Box2D.Position = Vector2.new(boxX, boxY)
                        else d.Box2D.Visible = false end

                        if Config.ESP_Box3D then
                            for _, l in pairs(d.Box3DLines) do l.Visible = true; l.Color = Config.ESP_Color end
                            d.Box3DLines[1].From = Vector2.new(boxX, boxY)
                            d.Box3DLines[1].To = Vector2.new(boxX + boxW, boxY)
                            d.Box3DLines[2].From = Vector2.new(boxX + boxW, boxY)
                            d.Box3DLines[2].To = Vector2.new(boxX + boxW, boxY + boxH)
                            d.Box3DLines[3].From = Vector2.new(boxX + boxW, boxY + boxH)
                            d.Box3DLines[3].To = Vector2.new(boxX, boxY + boxH)
                            d.Box3DLines[4].From = Vector2.new(boxX, boxY + boxH)
                            d.Box3DLines[4].To = Vector2.new(boxX, boxY)
                        else
                            for _, l in pairs(d.Box3DLines) do l.Visible = false end
                        end

                        if Config.ESP_SkeletonFF and (char:FindFirstChild("Head") and (char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso"))) then
                            local head = char:FindFirstChild("Head")
                            local torso = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
                            local la = char:FindFirstChild("LeftHand") or char:FindFirstChild("Left Arm")
                            local ra = char:FindFirstChild("RightHand") or char:FindFirstChild("Right Arm")
                            local ll = char:FindFirstChild("LeftFoot") or char:FindFirstChild("Left Leg")
                            local rl = char:FindFirstChild("RightFoot") or char:FindFirstChild("Right Leg")

                            for _, l in pairs(d.SkeletonLines) do l.Visible = true; l.Color = Config.ESP_Color end
                            
                            local function connectParts(p1, p2, lineIdx)
                                if p1 and p2 then
                                    local v1, s1 = Camera:WorldToViewportPoint(p1.Position)
                                    local v2, s2 = Camera:WorldToViewportPoint(p2.Position)
                                    if s1 and s2 then
                                        d.SkeletonLines[lineIdx].From = Vector2.new(v1.X, v1.Y)
                                        d.SkeletonLines[lineIdx].To = Vector2.new(v2.X, v2.Y)
                                    else
                                        d.SkeletonLines[lineIdx].Visible = false
                                    end
                                else
                                    d.SkeletonLines[lineIdx].Visible = false
                                end
                            end

                            connectParts(head, torso, 1)
                            connectParts(torso, la, 2)
                            connectParts(torso, ra, 3)
                            connectParts(torso, ll, 4)
                            connectParts(torso, rl, 5)
                            d.SkeletonLines[6].Visible = false
                        else
                            for _, l in pairs(d.SkeletonLines) do l.Visible = false end
                        end

                        if Config.ESP_Nickname then
                            d.Name.Visible = true
                            d.Name.Text = plr.Name
                            d.Name.Color = Config.ESP_Color
                            d.Name.Position = Vector2.new(vector.X, boxY - 14)
                        else d.Name.Visible = false end

                        if Config.ESP_Health then
                            d.HP.Visible = true
                            d.HP.Text = "HP: " .. math.floor(hum.Health)
                            d.HP.Color = Config.ESP_Color
                            d.HP.Position = Vector2.new(vector.X, boxY + boxH + 2)
                        else d.HP.Visible = false end

                        if Config.ESP_Line then
                            d.Tracer.Visible = true
                            d.Tracer.Color = Config.ESP_Color
                            d.Tracer.From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
                            d.Tracer.To = Vector2.new(vector.X, vector.Y)
                        else d.Tracer.Visible = false end
                    else
                        d.Box2D.Visible = false
                        for _, l in pairs(d.Box3DLines) do l.Visible = false end
                        for _, l in pairs(d.SkeletonLines) do l.Visible = false end
                        d.Name.Visible = false
                        d.HP.Visible = false
                        d.Tracer.Visible = false
                    end
                end
            end
        end
    end)
