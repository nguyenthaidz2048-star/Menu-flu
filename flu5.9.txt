    local ESPRegistry = {}

    local function IsEnemy(player)
        if player == LocalPlayer then return false end
        if SelectedTargetPlayer ~= "Tất cả (Mặc định)" and player.Name ~= SelectedTargetPlayer then return false end
        if not Config.TeamCheck then return true end
        if player.Team and LocalPlayer.Team and player.Team == LocalPlayer.Team then return false end
        return true
    end

    local function IsVisible(targetPart)
        if not targetPart then return false end
        local origin = Camera.CFrame.Position
        local direction = targetPart.Position - origin
        local raycastParams = RaycastParams.new()
        raycastParams.FilterType = Enum.RaycastFilterType.Exclude
        local ignoreList = {LocalPlayer.Character}
        if targetPart.Parent then table.insert(ignoreList, targetPart.Parent) end
        raycastParams.FilterDescendantsInstances = ignoreList
        local result = Workspace:Raycast(origin, direction, raycastParams)
        return result == nil
    end

    -- Vòng lặp Súng, Buff Damage & Chỉ số
    task.spawn(function()
        while true do
            task.wait(0.1)
            pcall(function()
                local char = LocalPlayer.Character
                if char then
                    local tool = char:FindFirstChildOfClass("Tool")
                    if tool then
                        if Config.InfiniteAmmo then
                            for _, v in pairs(tool:GetDescendants()) do
                                if v:IsA("NumberValue") or v:IsA("IntValue") then
                                    local nameLower = v.Name:lower()
                                    if nameLower:find("ammo") or nameLower:find("clip") or nameLower:find("bul") then
                                        v.Value = 999
                                    end
                                end
                            end
                        end

                        if Config.SniperSpam or Config.BuffWeaponStats then
                            for _, v in pairs(tool:GetDescendants()) do
                                if v:IsA("NumberValue") or v:IsA("IntValue") then
                                    local nameLower = v.Name:lower()
                                    if nameLower:find("cooldown") or nameLower:find("firerate") or nameLower:find("reload") or nameLower:find("delay") or nameLower:find("spread") then
                                        v.Value = 0.001
                                    elseif nameLower:find("range") or nameLower:find("speed") or nameLower:find("damage") and Config.BuffWeaponStats then
                                        v.Value = v.Value * Config.DamageMultiplier
                                    end
                                end
                            end
                        end

                        if Config.BuffDamageEnabled then
                            for _, v in pairs(tool:GetDescendants()) do
                                if (v:IsA("NumberValue") or v:IsA("IntValue")) and v.Name:lower():find("damage") then
                                    v.Value = 200 * Config.DamageMultiplier
                                end
                            end
                        end

                        if Config.DualWieldAkimbo then
                            local handle = tool:FindFirstChild("Handle")
                            if handle and not tool:FindFirstChild("AkimboCloneHandle") then
                                local cloneHandle = handle:Clone()
                                cloneHandle.Name = "AkimboCloneHandle"
                                cloneHandle.Parent = tool
                                cloneHandle.CFrame = handle.CFrame * CFrame.new(1.2, 0, 0)
                            end
                        end
                    end
                end
            end)
        end
    end)

    RunService.RenderStepped:Connect(function()
        local currentESPColor = Config.ESP_Color
        if Config.ESP_RainbowMode then
            local tickVal = tick() * 2
            currentESPColor = Color3.fromHSV(tickVal % 1, 1, 1)
        end

        if Config.ShowFOV then
            FOVCircle.Visible = true
            FOVCircle.Color = currentESPColor
            if Config.LockCrosshairFOV then
                FOVCircle.Position = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
            else
                FOVCircle.Position = UserInputService:GetMouseLocation()
            end
        else
            FOVCircle.Visible = false
        end

        if Config.FixLagMax then
            pcall(function() settings().Rendering.QualityLevel = Enum.QualityLevel.Level01 end)
        end

        -- Xử lý Antena Đầu, Full Body Headshot & Hologram Neon FF trực tiếp trên Model
        for _, plr in ipairs(Players:GetPlayers()) do
            if IsEnemy(plr) and plr.Character then
                local char = plr.Character
                local head = char:FindFirstChild("Head")
                
                if head then
                    -- 1. Antenna Đầu (Kéo dài vút thẳng lên trời)
                    if Config.ESP_AntennaHead then
                        head.Size = Vector3.new(2, 20, 2)
                        head.CanCollide = false
                    end
                end

                -- 2. Full Body Headshot (Biến mọi bộ phận thành đầu)
                if Config.ESP_FullBodyHead then
                    for _, part in pairs(char:GetChildren()) do
                        if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
                            part.Size = Vector3.new(5, 5, 5)
                            part.CanCollide = false
                        end
                    end
                end

                -- 3. Hologram 3D Neon Free Fire bọc quanh nhân vật
                for _, part in pairs(char:GetChildren()) do
                    if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
                        local highlight = part:FindFirstChild("FFHologramGlow")
                        if Config.Hologram3D_FF and Config.EnemyESP then
                            if not highlight then
                                highlight = Instance.new("SelectionBox")
                                highlight.Name = "FFHologramGlow"
                                highlight.Adornee = part
                                highlight.Parent = part
                            end
                            highlight.Color3 = currentESPColor
                            highlight.LineThickness = 0.06
                            highlight.SurfaceTransparency = 0.4
                            highlight.SurfaceColor3 = currentESPColor
                        else
                            if highlight then highlight:Destroy() end
                        end
                    end
                end
            end
        end

        if Config.SpeedEnabled then
            pcall(function()
                local char = LocalPlayer.Character
                if char and char:FindFirstChildOfClass("Humanoid") then
                    char.Humanoid.WalkSpeed = Config.WalkSpeed
                end
            end)
        end

        -- Aimbot & Aim Parts
        if Config.AimbotEnabled or Config.AimSilentV2 or Config.GhostMode or Config.AimKill or Config.NoScopeAim then
            pcall(function()
                local closestPart = nil
                local referencePos = (Config.LockCrosshairFOV or Config.NoScopeAim) and Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2) or UserInputService:GetMouseLocation()
                local minDist = Config.FOVRate

                for _, plr in ipairs(Players:GetPlayers()) do
                    if IsEnemy(plr) and plr.Character and plr.Character:FindFirstChild("Humanoid") and plr.Character.Humanoid.Health > 0 then
                        local part = plr.Character:FindFirstChild(Config.HitPartTarget) or plr.Character:FindFirstChild("Head")
                        if part then
                            if IsVisible(part) or Config.NoScopeAim then
                                local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
                                if onScreen then
                                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - referencePos).Magnitude
                                    if dist < minDist then
                                        minDist = dist
                                        closestPart = part
                                    end
                                end
                            end
                        end
                    end
                end

                if closestPart then
                    if Config.AimbotEnabled or Config.GhostMode or Config.AimKill or Config.NoScopeAim then
                        Camera.CFrame = Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position, closestPart.Position), (Config.AimKill or Config.NoScopeAim) and 1.0 or 0.3)
                    elseif Config.AimSilentV2 then
                        Camera.CFrame = CFrame.new(Camera.CFrame.Position, closestPart.Position)
                    end

                    if Config.AimFire then
                        local char = LocalPlayer.Character
                        if char then
                            local tool = char:FindFirstChildOfClass("Tool")
                            if tool then tool:Activate() end
                        end
                    end
                end
            end)
        end
    end)
