            stateLbl.Text = val and "ON" or "OFF"
            stateLbl.TextColor3 = val and Color3.fromRGB(150, 50, 255) or Color3.fromRGB(255, 70, 70)
            cb(val)
        end)

        btn.Position = UDim2.new(0.02, 0, 0, yPos + 18)

    AddToggle(TabMovement, "Speed (Chạy nhanh)", Config.SpeedEnabled, 35, "VIP", function(v) Config.SpeedEnabled = v end)
    AddSlider(TabMovement, "WalkSpeed Value", 16, 150, Config.WalkSpeed, 65, "VIP", function(v) Config.WalkSpeed = v end)
    AddToggle(TabMovement, "Cam Cao (Zoom)", Config.CamHighEnabled, 100, "Thuong", function(v) Config.CamHighEnabled = v end)
    AddSlider(TabMovement, "Camera Distance", 10, 150, Config.CamDistance, 130, "Thuong", function(v) Config.CamDistance = v end)
    AddToggle(TabMovement, "Fly Cam Real", Config.FlyCamReal, 165, "Full", function(v) Config.FlyCamReal = v end)

    -- Tab Misc (Bảo vệ Anti-Ban / Anti-Kick luôn có sẵn cho mọi loại key)
    AddToggle(TabMisc, "Anti-Ban (Bảo vệ)", Config.AntiBan, 5, "Thuong", function(v) Config.AntiBan = v end)
    AddToggle(TabMisc, "Anti-Kick (Chống văng)", Config.AntiKick, 35, "Thuong", function(v) Config.AntiKick = v end)
    AddToggle(TabMisc, "Fix Lag Max (Tối ưu)", Config.FixLagMax, 65, "Thuong", function(v) Config.FixLagMax = v end)
    AddToggle(TabMisc, "Ghost Mode", Config.GhostMode, 95, "VIP", function(v) Config.GhostMode = v end)

    -- ========================================================================
    -- HỆ THỐNG XỬ LÝ LOGIC (RUNSERVICE)
    -- ========================================================================
    local ESPRegistry = {}
    local flyPos = Vector3.new(0, 50, 0)

    local function IsEnemy(player)
        if player == LocalPlayer then return false end
        if not Config.TeamCheck then return true end
        if player.Team and LocalPlayer.Team and player.Team == LocalPlayer.Team then return false end
        return true
    end

    -- Hook Anti-Kick & Anti-Ban
    pcall(function()
        if Config.AntiKick and getrawmetatable then
            local mt = getrawmetatable(game)
            local old = mt.__namecall
            setreadonly(mt, false)
            mt.__namecall = newcclosure(function(self, ...)
                local method = getnamecallmethod()
                if method == "Kick" and self == LocalPlayer then return end
                return old(self, ...)
            end)
            setreadonly(mt, true)
        end
    end)

    RunService.RenderStepped:Connect(function(dt)
        -- FOV Circle
        if Config.ShowFOV then
            FOVCircle.Visible = true
            if Config.LockCrosshairFOV then
                FOVCircle.Position = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
            else
                FOVCircle.Position = UserInputService:GetMouseLocation()
            end
        else
            FOVCircle.Visible = false
        end

        -- Fix Lag Max
        if Config.FixLagMax then
            pcall(function()
                settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
                for _, obj in pairs(Workspace:GetDescendants()) do
                    if obj:IsA("ParticleEmitter") or obj:IsA("Trail") then obj.Enabled = false end
                end
            end)
        end

        -- NoClip & Speed
        if Config.NoClip then
            pcall(function()
                local char = LocalPlayer.Character
                if char then
                    for _, p in pairs(char:GetDescendants()) do if p:IsA("BasePart") then p.CanCollide = false end end
                end
            end)
        end

        if Config.SpeedEnabled then
            pcall(function()
                local char = LocalPlayer.Character
                if char and char:FindFirstChildOfClass("Humanoid") then
                    char.Humanoid.WalkSpeed = Config.WalkSpeed
                end
            end)
        end

        -- Rapid Fire & Fast Reload (VIP / Full)
        if Config.RapidFireEnabled then
            pcall(function()
                local char = LocalPlayer.Character
                if char then
                    local tool = char:FindFirstChildOfClass("Tool")
                    if tool and UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then
                        task.wait(1 / Config.BulletPerSecond)
                        pcall(function() tool:Activate() end)
                    end
                end
            end)
        end

        if Config.FastReload then
            pcall(function()
                local char = LocalPlayer.Character
                if char then
                    local tool = char:FindFirstChildOfClass("Tool")
                    if tool then
                        -- Tự động tinh chỉnh các thông số thời gian thay đạn nếu tool hỗ trợ
                        for _, v in pairs(tool:GetDescendants()) do
                            if v:IsA("NumberValue") and (v.Name:lower():find("reload") or v.Name:lower():find("cooldown")) then
                                v.Value = 0.05
                            end
                        end
                    end
                end
            end)
        end

        -- Aimbot Logic
        if Config.AimbotEnabled or Config.AimSilentV2 then
            pcall(function()
                local closestPart = nil
                local referencePos = Config.LockCrosshairFOV and Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2) or UserInputService:GetMouseLocation()
                local minDist = Config.FOVRate

                for _, plr in ipairs(Players:GetPlayers()) do
                    if IsEnemy(plr) and plr.Character and plr.Character:FindFirstChild("Humanoid") and plr.Character.Humanoid.Health > 0 then
                        local part = plr.Character:FindFirstChild("Head")
                        if part then
                            local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
                            if onScreen then
                                local dist = (Vector2.new(screenPos.X, screenPos.Y) - referencePos).Magnitude
                                if dist < minDist then
                                    minDist = dist
                                    closestPart = part
                                end
                            end
                        end
                    end
                end

                if closestPart then
                    local smoothFactor = (Config.AimSpeedMode == "Chậm" and 12.0) or (Config.AimSpeedMode == "Khóa tâm" and 1.0) or 3.0
                    if Config.AimbotEnabled then
                        Camera.CFrame = Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position, closestPart.Position), 1 / smoothFactor)
                    elseif Config.AimSilentV2 then
                        Camera.CFrame = CFrame.new(Camera.CFrame.Position, closestPart.Position)
                    end
                end
            end)
        end

        -- Kill Aura 700m
        if Config.KillAura700m then
            pcall(function()
                local myChar = LocalPlayer.Character
                if myChar and myChar:FindFirstChild("HumanoidRootPart") then
                    local myPos = myChar.HumanoidRootPart.Position
                    for _, plr in ipairs(Players:GetPlayers()) do
                        if IsEnemy(plr) and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                            local enemyRoot = plr.Character.HumanoidRootPart
                            if (enemyRoot.Position - myPos).Magnitude <= 700 then
                                pcall(function()
                                    if myChar:FindFirstChildOfClass("Tool") then myChar:FindFirstChildOfClass("Tool"):Activate() end
                                end)
                            end
                        end
                    end
                end
            end)
        end

        -- ESP & Skeleton Rendering
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer then
                local char = plr.Character
                if char and char:FindFirstChild("HumanoidRootPart") and char:FindFirstChildOfClass("Humanoid") then
                    local hrp = char.HumanoidRootPart
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    local head = char:FindFirstChild("Head")

                    if not ESPRegistry[plr] then
                        ESPRegistry[plr] = {
                            Box = Drawing.new("Square"),
                            Name = Drawing.new("Text"),
                            HP = Drawing.new("Text"),
                            Tracer = Drawing.new("Line"),
                            HeadDot = Drawing.new("Circle"),
                            SkeletonLines = { Drawing.new("Line"), Drawing.new("Line"), Drawing.new("Line"), Drawing.new("Line"), Drawing.new("Line") }
                        }
                        local d = ESPRegistry[plr]
                        d.Box.Filled = false
                        d.Name.Center = true
                        d.Name.Size = 11
                        d.HP.Center = true
                        d.HP.Size = 10
                        d.HeadDot.Filled = true
                        d.HeadDot.Radius = 4
                    end

                    local d = ESPRegistry[plr]
                    local vector, onScreen = Camera:WorldToViewportPoint(hrp.Position)

                    if onScreen and hum.Health > 0 and Config.EnemyESP then
                        local boxW = 1500 / vector.Z
                        local boxH = 2600 / vector.Z
                        local boxX = vector.X - boxW / 2
                        local boxY = vector.Y - boxH / 2

                        if Config.ESP_Box then
                            d.Box.Visible = true
                            d.Box.Color = Config.ESP_Color
                            d.Box.Size = Vector2.new(boxW, boxH)
                            d.Box.Position = Vector2.new(boxX, boxY)
                        else d.Box.Visible = false end

                        if Config.ESP_Nickname then
                            d.Name.Visible = true
                            d.Name.Text = plr.Name
                            d.Name.Color = Config.ESP_Color
                            d.Name.Position = Vector2.new(vector.X, boxY - 14)
                        else d.Name.Visible = false end

                        if Config.ESP_Health then
                            d.HP.Visible = true
                            d.HP.Text = "HP: " .. math.floor(hum.Health)
                            d.HP.Color = Config.ESP_Color
                            d.HP.Position = Vector2.new(vector.X, boxY + boxH + 2)
                        else d.HP.Visible = false end

                        if Config.ESP_Line then
                            d.Tracer.Visible = true
                            d.Tracer.Color = Config.ESP_Color
                            d.Tracer.From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
                            d.Tracer.To = Vector2.new(vector.X, vector.Y)
                        else d.Tracer.Visible = false end

                        -- Skeleton ESP (Chỉ hiển thị nếu bật và có quyền VIP/Full)
                        if Config.ESP_Skeleton and UserAccessLevel ~= "Thuong" then
                            pcall(function()
                                local torso = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
                                if head and torso then
                                    local hPos, hVis = Camera:WorldToViewportPoint(head.Position)
                                    local tPos, tVis = Camera:WorldToViewportPoint(torso.Position)
                                    if hVis and tVis then
                                        d.SkeletonLines[1].Visible = true
                                        d.SkeletonLines[1].Color = Config.ESP_Color
                                        d.SkeletonLines[1].From = Vector2.new(hPos.X, hPos.Y)
                                        d.SkeletonLines[1].To = Vector2.new(tPos.X, tPos.Y)
                                    else d.SkeletonLines[1].Visible = false end
                                else
                                    for _, l in ipairs(d.SkeletonLines) do l.Visible = false end
                                end
                            end)
                        else
                            for _, l in ipairs(d.SkeletonLines) do l.Visible = false end
                        end
                    else
                        for _, obj in pairs(d) do 
                            if typeof(obj) == "table" then for _, l in ipairs(obj) do l.Visible = false end
                            else obj.Visible = false end
                        end
                    end
                end
            end
        end
    end)

    print("[FLU V3] Ultimate Hub Loaded Successfully with Access Level: " .. UserAccessLevel)
end
