local function CreateTab(tabName, iconText)

    page.Size = UDim2.new(1, 0, 1, 0)
    page.CanvasSize = UDim2.new(0, 0, 3.2, 0)
    page.ScrollBarThickness = 2
    page.Visible = false

    local btn = Instance.new("TextButton")
    btn.Parent = Sidebar
    btn.BackgroundColor3 = Color3.fromRGB(22, 18, 32)
    btn.Size = UDim2.new(0.9, 0, 0, 28)
    btn.Position = UDim2.new(0.05, 0, 0, (#TabButtons * 32) + 6)
    btn.Font = Enum.Font.GothamBold

    btn.TextColor3 = Color3.fromRGB(160, 140, 190)
    btn.TextSize = 10
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 4)

    btn.MouseButton1Click:Connect(function()
        for _, p in pairs(Tabs) do p.Visible = false end
        for _, b in pairs(TabButtons) do 
            b.BackgroundColor3 = Color3.fromRGB(22, 18, 32)
            b.TextColor3 = Color3.fromRGB(160, 140, 190)
        end
        page.Visible = true
        btn.BackgroundColor3 = Color3.fromRGB(130, 30, 220)
        btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    end)

    table.insert(TabButtons, btn)
    Tabs[tabName] = page
    return page
end

local function AddToggle(parent, text, defaultVal, yPos, cb)

    btn.Parent = parent
    btn.BackgroundColor3 = Color3.fromRGB(25, 21, 36)
    btn.Position = UDim2.new(0.02, 0, 0, yPos)

    local stateLbl = Instance.new("TextLabel")
    stateLbl.Parent = btn
    stateLbl.BackgroundTransparency = 1
    stateLbl.Position = UDim2.new(1, -55, 0, 0)
    stateLbl.Size = UDim2.new(0, 50, 1, 0)
    stateLbl.Font = Enum.Font.GothamBold

    stateLbl.TextColor3 = defaultVal and Color3.fromRGB(150, 50, 255) or Color3.fromRGB(255, 70, 70)

    lbl.TextColor3 = Color3.fromRGB(200, 190, 215)
    lbl.TextSize = 10
    lbl.TextXAlignment = Enum.TextXAlignment.Left

    local box = Instance.new("TextBox")
    box.Parent = parent
    box.BackgroundColor3 = Color3.fromRGB(28, 23, 40)
    box.Position = UDim2.new(0.65, 0, 0, yPos)
    box.Size = UDim2.new(0.33, 0, 0, 24)
    box.Font = Enum.Font.GothamBold
    box.Text = tostring(defaultVal)
    box.TextColor3 = Color3.fromRGB(190, 100, 255)
    box.TextSize = 10
    Instance.new("UICorner", box).CornerRadius = UDim.new(0, 4)

    box.FocusLost:Connect(function()
        local n = tonumber(box.Text)
        if n then
            local clamped = math.clamp(n, minVal, maxVal)
            box.Text = tostring(clamped)
            cb(clamped)
        end
    end)
end

local ESPRegistry = {}
local flyPos = Vector3.new(0, 50, 0)

-- Hàm kiểm tra Team chuẩn xác
local function IsEnemy(player)
    if player == LocalPlayer then return false end
    if not Config.TeamCheck then return true end
    
    if player.Team and LocalPlayer.Team then
        if player.Team == LocalPlayer.Team then return false end
    end
    
    if player.TeamColor and LocalPlayer.TeamColor then
        if player.TeamColor == LocalPlayer.TeamColor then return false end
    end
    
    return true
end

-- Anti Kick & Ban Hook Cơ Bản
pcall(function()
    if Config.AntiKick and getrawmetatable then
        local mt = getrawmetatable(game)
        local old = mt.__namecall
        setreadonly(mt, false)
        mt.__namecall = newcclosure(function(self, ...)
            local method = getnamecallmethod()
            if method == "Kick" and self == LocalPlayer then return end
            return old(self, ...)
        end)
        setreadonly(mt, true)
    end
end)

RunService.RenderStepped:Connect(function(dt)
    -- FOV Circle
    if Config.ShowFOV then
        FOVCircle.Visible = true
        if Config.LockCrosshairFOV then
            FOVCircle.Position = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
        else
            FOVCircle.Position = UserInputService:GetMouseLocation()
        end
    else
        FOVCircle.Visible = false
    end

    -- 1. Fix Lag Max
    if Config.FixLagMax then
        pcall(function()
            settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
            for _, obj in pairs(Workspace:GetDescendants()) do
                if obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Fire") or obj:IsA("Sparkles") then
                    obj.Enabled = false
                end
            end
        end)
    end

    -- 2. NoClip
    if Config.NoClip then
        pcall(function()
            local char = LocalPlayer.Character
            if char then
                for _, part in pairs(char:GetDescendants()) do
                    if part:IsA("BasePart") then part.CanCollide = false end
                end
            end
        end)
    end

    -- 3. Speed
    if Config.SpeedEnabled then
        pcall(function()
            local char = LocalPlayer.Character
            if char and char:FindFirstChildOfClass("Humanoid") then
                char.Humanoid.WalkSpeed = Config.WalkSpeed
            end
        end)
    end

    -- 4. Cam Cao & Góc Nhìn
    if Config.CamHighEnabled then
        pcall(function()
            LocalPlayer.CameraMaxZoomDistance = Config.CamDistance
            LocalPlayer.CameraMinZoomDistance = Config.CamDistance
        end)
    else
        pcall(function()
            LocalPlayer.CameraMaxZoomDistance = 400
            LocalPlayer.CameraMinZoomDistance = 0.5
        end)
    end

    pcall(function()
        if Config.CameraPerspective == "Góc Nhìn 1" then
            LocalPlayer.CameraMode = Enum.CameraMode.LockFirstPerson
        elseif Config.CameraPerspective == "Góc Nhìn 2" then
            LocalPlayer.CameraMode = Enum.CameraMode.Classic
            LocalPlayer.CameraMinZoomDistance = 2
            LocalPlayer.CameraMaxZoomDistance = 6
        else
            LocalPlayer.CameraMode = Enum.CameraMode.Classic
            LocalPlayer.CameraMinZoomDistance = 10
            LocalPlayer.CameraMaxZoomDistance = 50
        end
    end)

    -- 5. Fly Cam Real
    if Config.FlyCamReal then
        pcall(function()
            Camera.CameraType = Enum.CameraType.Scriptable
            local moveDir = Vector3.new()
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + Camera.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - Camera.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - Camera.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + Camera.CFrame.RightVector end
            flyPos = flyPos + (moveDir * 60 * dt)
            Camera.CFrame = CFrame.new(flyPos, flyPos + Camera.CFrame.LookVector)
        end)
    else
        if Camera.CameraType == Enum.CameraType.Scriptable and not Config.FlyCamReal then
            Camera.CameraType = Enum.CameraType.Custom
        end
    end

    -- 6. Hitbox Expander
    if Config.HitboxExpander then
        pcall(function()
            for _, plr in ipairs(Players:GetPlayers()) do
                if IsEnemy(plr) and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                    local hrp = plr.Character.HumanoidRootPart
                    hrp.Size = Vector3.new(Config.HitboxSize, Config.HitboxSize, Config.HitboxSize)
                    hrp.Transparency = 0.7
                    hrp.BrickColor = BrickColor.new("Magenta")
                    hrp.Material = Enum.Material.Neon
                    hrp.CanCollide = false
                end
            end
        end)
    end

    -- 7. Rapid Fire
    if Config.RapidFireEnabled then
        pcall(function()
            local char = LocalPlayer.Character
            if char then
                local tool = char:FindFirstChildOfClass("Tool")
                if tool and UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then
                    task.wait(1 / Config.BulletPerSecond)
                    pcall(function() tool:Activate() end)
                end
            end
        end)
    end

    -- 8. Aimbot & Silent Aim V2 (Fix lệch tâm bằng cách nhắm chuẩn các phần cơ thể R15/R6)
    if Config.AimbotEnabled or Config.AimSilentV2 then
        pcall(function()
            local targetPartName = "Head"
            if Config.HitPartTarget == "Neck" then targetPartName = "UpperTorso"
            elseif Config.HitPartTarget == "Stomach" then targetPartName = "LowerTorso"
            elseif Config.HitPartTarget == "Normal" then targetPartName = "HumanoidRootPart" end

            local closestPart = nil
            local referencePos = Config.LockCrosshairFOV and Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2) or UserInputService:GetMouseLocation()
            local minDist = Config.FOVRate

            for _, plr in ipairs(Players:GetPlayers()) do
                if IsEnemy(plr) and plr.Character and plr.Character:FindFirstChild("Humanoid") and plr.Character.Humanoid.Health > 0 then
                    local part = plr.Character:FindFirstChild(targetPartName) or plr.Character:FindFirstChild("Head")
                    if part then
                        local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
                        if onScreen then
                            local dist = (Vector2.new(screenPos.X, screenPos.Y) - referencePos).Magnitude
                            if dist < minDist then
                                minDist = dist
                                closestPart = part
                            end
                        end
                    end
                end
            end

            if closestPart then
                local smoothFactor = 3.0
                if Config.AimSpeedMode == "Chậm" then
                    smoothFactor = 12.0
                elseif Config.AimSpeedMode == "Khóa tâm" then
                    smoothFactor = 1.0
                end

                if Config.AimbotEnabled then
                    local targetCFrame = CFrame.new(Camera.CFrame.Position, closestPart.Position)
                    Camera.CFrame = Camera.CFrame:Lerp(targetCFrame, 1 / smoothFactor)
                elseif Config.AimSilentV2 then
                    local currentCam = Camera.CFrame
                    Camera.CFrame = CFrame.new(currentCam.Position, closestPart.Position)
                end
            end
        end)
    end

    -- 9. Kill Aura 700m
    if Config.KillAura700m then

                                if myChar:FindFirstChildOfClass("Tool") then

                local hum = char:FindFirstChildOfClass("Humanoid")
                local head = char:FindFirstChild("Head")

                if not ESPRegistry[plr] then
                    ESPRegistry[plr] = {
                        Box = Drawing.new("Square"),
                        Name = Drawing.new("Text"),
                        HP = Drawing.new("Text"),
                        Tracer = Drawing.new("Line"),
                        HeadDot = Drawing.new("Circle"),
                        SkeletonLines = {
                            Drawing.new("Line"), -- Đầu -> Cổ/Ngực
                            Drawing.new("Line"), -- Ngực -> Tay trái
                            Drawing.new("Line"), -- Ngực -> Tay phải

                            Drawing.new("Line"), -- Hông -> Chân trái

                    local d = ESPRegistry[plr]
                    d.Box.Filled = false
                    d.Box.Thickness = 1

                    d.Name.Size = 11

                    d.HeadDot.Radius = 4
                    for _, l in ipairs(d.SkeletonLines) do
                        l.Thickness = 1.2

                local d = ESPRegistry[plr]

                            d.HeadDot.Visible = true
                            d.HeadDot.Color = Config.ESP_Color
                            d.HeadDot.Position = Vector2.new(headVec.X, headVec.Y)
                        else d.HeadDot.Visible = false end
                    else d.HeadDot.Visible = false end

                    -- Vẽ Skeleton (Khung xương chi tiết chống lệch tâm)
                    if Config.ESP_Skeleton then
                        pcall(function()
                            local torso = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
                            local leftArm = char:FindFirstChild("LeftUpperArm") or char:FindFirstChild("Left Arm")
                            local rightArm = char:FindFirstChild("RightUpperArm") or char:FindFirstChild("Right Arm")

                                    d.SkeletonLines[1].Visible = true
                                    d.SkeletonLines[1].Color = Config.ESP_Color
                                    d.SkeletonLines[1].From = Vector2.new(hPos.X, hPos.Y)

                                else d.SkeletonLines[1].Visible = false end

                                if leftArm then
                                    local laPos, laVis = Camera:WorldToViewportPoint(leftArm.Position)

                                        d.SkeletonLines[2].Color = Config.ESP_Color
                                        d.SkeletonLines[2].From = Vector2.new(tPos.X, tPos.Y)
                                        d.SkeletonLines[2].To = Vector2.new(laPos.X, laPos.Y)
                                    else d.SkeletonLines[2].Visible = false end
                                else d.SkeletonLines[2].Visible = false end

                                if rightArm then
                                    local raPos, raVis = Camera:WorldToViewportPoint(rightArm.Position)
                                    if raVis then

                                        d.SkeletonLines[3].Color = Config.ESP_Color
                                        d.SkeletonLines[3].From = Vector2.new(tPos.X, tPos.Y)

                                else d.SkeletonLines[3].Visible = false end

                                    if llVis then
                                        d.SkeletonLines[5].Visible = true
                                        d.SkeletonLines[5].Color = Config.ESP_Color
                                        d.SkeletonLines[5].From = Vector2.new(tPos.X, tPos.Y)
                                        d.SkeletonLines[5].To = Vector2.new(llPos.X, llPos.Y)
                                    else d.SkeletonLines[5].Visible = false end
                                else d.SkeletonLines[5].Visible = false end

                                    local rlPos, rlVis = Camera:WorldToViewportPoint(rightLeg.Position)
                                    if rlVis then
                                        d.SkeletonLines[6].Visible = true
                                        d.SkeletonLines[6].Color = Config.ESP_Color
                                        d.SkeletonLines[6].From = Vector2.new(tPos.X, tPos.Y)
                                        d.SkeletonLines[6].To = Vector2.new(rlPos.X, rlPos.Y)

                                else d.SkeletonLines[6].Visible = false end
                            else
                                for _, l in ipairs(d.SkeletonLines) do l.Visible = false end
                            end
                        end)
                    else
                        for _, l in ipairs(d.SkeletonLines) do l.Visible = false end
                    end
                else
                    for _, obj in pairs(d) do 
                        if typeof(obj) == "table" then
                            for _, l in ipairs(obj) do l.Visible = false end
                        else
                            obj.Visible = false 
                        end
                    end

            end
        end
